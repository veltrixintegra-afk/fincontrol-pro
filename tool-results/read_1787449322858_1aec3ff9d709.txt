    68→// ===== Custom Recharts Tooltip =====
    69→function CustomTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color?: string }>; label?: string }) {
    70→  if (!active || !payload?.length) return null;
    71→  return (
    72→    <div className="rounded-lg border border-slate-600 bg-slate-800 px-3 py-2 text-xs shadow-xl">
    73→      {label && <p className="mb-1 font-medium text-slate-300">{label}</p>}
    74→      {payload.map((entry, i) => (
    75→        <div key={i} className="flex items-center gap-2">
    76→          <span className="h-2 w-2 rounded-sm" style={{ backgroundColor: entry.color }} />
    77→          <span className="text-slate-400">{entry.name}:</span>
    78→          <span className="font-mono font-medium text-white">{formatCLP(entry.value)}</span>
    79→        </div>
    80→      ))}
    81→    </div>
    82→  );
    83→}

   192→// ===== Donut Chart =====
   193→function DonutChartSection({ data, vistaActual, onSectorClick }: {
   194→  data: ReturnType<typeof calcularTodo>;
   195→  vistaActual: ViewType;
   196→  onSectorClick: (key: string) => void;
   197→}) {
   198→  const { labels, values, colors, title } = useMemo(() => {
   199→    if (vistaActual === 'generales') {
   200→      const keys: CategoryKey[] = ['mayores', 'menores', 'inversiones'];
   201→      const lbls = keys.map(k => `${CATEGORIAS[k].nombre} (${CATEGORIAS[k].pctTotal}%)`);
   202→      const vals = keys.map(k => data.categorias[k].monto);
   203→      const cols = keys.map(k => CATEGORIAS[k].color);
   204→      if (data.sinAsignar > 0) {
   205→        lbls.push('Sin Asignar');
   206→        vals.push(data.sinAsignar);
   207→        cols.push('#64748b');
   208→      }
   209→      return { labels: lbls, values: vals, colors: cols, title: 'Distribución del Ingreso' };
   210→    }
   211→    const cat = CATEGORIAS[vistaActual as CategoryKey];
   212→    const activos = data.categorias[vistaActual as CategoryKey].conceptos.filter(c => c.activo);
   213→    return {
   214→      labels: activos.map(c => c.nombre),
   215→      values: activos.map(c => c.monto),
   216→      colors: activos.map(c => c.color),
   217→      title: `Desglose: ${cat.nombre}`,
   218→    };
   219→  }, [data, vistaActual]);
   220→
   221→  const chartData = labels.map((name, i) => ({ name, value: values[i], fill: colors[i] }));
   222→
   223→  return (
   224→    <Card className="bg-slate-800/80 border border-slate-700/50">
   225→      <CardHeader className="pb-2">
   226→        <CardTitle className="text-base font-bold text-slate-100">{title}</CardTitle>
   227→      </CardHeader>
   228→      <CardContent>
   229→        <div className="h-[300px]">
   230→          <ResponsiveContainer width="100%" height="100%">
   231→            <PieChart>
   232→              <Pie
   233→                data={chartData}
   234→                cx="50%"
   235→                cy="50%"
   236→                innerRadius={70}
   237→                outerRadius={110}
   238→                paddingAngle={2}
   239→                dataKey="value"
   240→                cursor="pointer"
   241→                onClick={(_, index) => {
   242→                  if (vistaActual === 'generales') {
   243→                    const keys: CategoryKey[] = ['mayores', 'menores', 'inversiones'];
   244→                    if (index < keys.length) onSectorClick(keys[index]);
   245→                  }
   246→                }}
   247→              >
   248→                {chartData.map((entry, index) => (
   249→                  <Cell key={`cell-${index}`} fill={entry.fill} stroke="transparent" />
   250→                ))}
   251→              </Pie>
   252→              <Tooltip content={<CustomTooltip />} />
   253→            </PieChart>
   254→          </ResponsiveContainer>
   255→        </div>
   256→      </CardContent>
   257→    </Card>
   258→  );
   259→}
